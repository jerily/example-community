cmake_minimum_required(VERSION 3.16)
project(example-community VERSION 0.1.7 LANGUAGES NONE)
set(PACKAGE_DIR "lib/example-community${PROJECT_VERSION}")
install(FILES pkgIndex.tcl readme.md config.ini LICENSE start.sh DESTINATION "${PACKAGE_DIR}")
install(DIRECTORY library assets DESTINATION "${PACKAGE_DIR}")
include(CTest)
if(BUILD_TESTING)
    find_program(TCLSH NAMES tclsh9.1 tclsh REQUIRED)
    add_test(NAME example-community COMMAND "${TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl")
endif()

install(FILES main.tcl DESTINATION "${PACKAGE_DIR}")
install(DIRECTORY content DESTINATION "${PACKAGE_DIR}")
install(PROGRAMS bin/example-community DESTINATION bin)
